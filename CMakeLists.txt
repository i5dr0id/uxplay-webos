cmake_minimum_required(VERSION 3.16)
project(uxplay_webos LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED sdl2)
pkg_check_modules(TTF REQUIRED SDL2_ttf)
pkg_check_modules(SMP REQUIRED libplayerAPIs)
add_library(lg_utils STATIC src/utils.cpp)
target_include_directories(lg_utils PUBLIC src)
add_library(lg_media STATIC src/media.cpp src/shell.cpp)
target_include_directories(lg_media PUBLIC src ${SDL_INCLUDE_DIRS} ${TTF_INCLUDE_DIRS} ${SMP_INCLUDE_DIRS})
target_link_directories(lg_media PUBLIC ${SDL_LIBRARY_DIRS} ${SMP_LIBRARY_DIRS})
target_link_libraries(lg_media PUBLIC lg_utils ${SDL_LIBRARIES} ${TTF_LIBRARIES} ${SMP_LIBRARIES} pthread)
target_compile_options(lg_media PRIVATE -Wall -Wextra)
add_executable(media-probe src/probe.cpp)
target_link_libraries(media-probe PRIVATE lg_media)
set_target_properties(media-probe PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
option(BUILD_RECEIVER "Build the AirPlay receiver (requires tools/build-deps.sh)" OFF)
if(BUILD_RECEIVER)
    set(DEPS "${CMAKE_SOURCE_DIR}/.cache/deps" CACHE PATH "Private ARM dependencies")
    set(UX "${CMAKE_SOURCE_DIR}/vendor/uxplay")
    add_subdirectory(${UX}/lib/llhttp)
    add_subdirectory(${UX}/lib/playfair)
    add_subdirectory(${UX}/lib/mdnsd)
    file(GLOB CORE CONFIGURE_DEPENDS "${UX}/lib/*.c")
    add_library(airplay STATIC ${CORE})
    target_compile_definitions(airplay PRIVATE PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L _REENTRANT _LARGEFILE64_SOURCE _FILE_OFFSET_BITS=64)
    target_include_directories(airplay PUBLIC "${UX}/lib" PRIVATE "${UX}/lib/llhttp" "${UX}/lib/playfair" "${DEPS}/include")
    target_link_libraries(airplay PUBLIC playfair dnssd llhttp "${DEPS}/lib/libplist-2.0.a" "${DEPS}/lib/libcrypto.a" pthread dl m)
    add_executable(uxplay-webos src/main.cpp src/receiver.cpp src/audio.cpp src/h264.cpp)
    target_include_directories(uxplay-webos PRIVATE "${DEPS}/include")
    target_link_libraries(uxplay-webos PRIVATE lg_media airplay "${DEPS}/lib/libavcodec.a" "${DEPS}/lib/libswresample.a" "${DEPS}/lib/libavutil.a")
    target_link_options(uxplay-webos PRIVATE -Wl,--exclude-libs,ALL)
    set_target_properties(uxplay-webos PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
    add_executable(uxplay-daemon src/daemon.cpp src/receiver.cpp)
    target_include_directories(uxplay-daemon PRIVATE "${DEPS}/include")
    target_link_libraries(uxplay-daemon PRIVATE lg_utils airplay)
    target_link_options(uxplay-daemon PRIVATE -Wl,--exclude-libs,ALL)
    set_target_properties(uxplay-daemon PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
